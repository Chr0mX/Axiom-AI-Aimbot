# TensorRT backend dependencies
#
# Recommended — use the installer (handles Python version detection automatically):
#   python src\install_tensorrt_local.py
#
# Or manually (install sub-packages directly to avoid Python 3.13+ build failure):
#   pip install tensorrt_cu12_libs<11 ^
#       --target "%LOCALAPPDATA%\AxiomAI\site-packages" ^
#       --extra-index-url https://pypi.nvidia.com
#   # Python <= 3.12 only — add bindings:
#   pip install tensorrt_cu12_bindings<11 ^
#       --target "%LOCALAPPDATA%\AxiomAI\site-packages" ^
#       --extra-index-url https://pypi.nvidia.com
#
# Compatibility matrix
# ─────────────────────────────────────────────────────────────────────────────
# CUDA toolkit : 12.x  (driver >= 525.x)
# cuDNN        : 9.x   (bundled in nvidia-cudnn-cu12)
# TensorRT     : 10.x  (tensorrt_cu12_libs < 11)
# ORT          : >= 1.19 (required for TRT 10 EP)
# Python       : 3.10 – 3.14  (bindings only on 3.10–3.12; DLLs work on all)
# OS           : Windows 10/11 x64
# ─────────────────────────────────────────────────────────────────────────────

# ── ONNX Runtime (CUDA + TensorRT execution providers) ───────────────────────
onnxruntime-gpu

# ── NVIDIA CUDA runtime wheels (from standard PyPI) ──────────────────────────
nvidia-cublas-cu12
nvidia-cuda-runtime-cu12
nvidia-cudnn-cu12
nvidia-cufft-cu12
nvidia-curand-cu12
nvidia-cusolver-cu12
nvidia-cusparse-cu12

# ── TensorRT 10 DLLs + Python bindings (from pypi.nvidia.com) ────────────────
# Install the sub-packages directly — the tensorrt-cu12 meta-package fails on
# Python > 3.12 because tensorrt_cu12_bindings has no wheel for those versions.
#
#   tensorrt_cu12_libs     — nvinfer_10.dll, nvonnxparser_10.dll (any Python)
#   tensorrt_cu12_bindings — `import tensorrt` Python API (Python ≤ 3.12 only)
#
# ORT's TensorrtExecutionProvider only needs the DLLs; bindings are optional.
# Pinned to <11: ORT links nvinfer_10.dll; TRT 11 ships nvinfer_11.dll, which
# makes the TRT EP fail with "nvinfer_10.dll missing".
--extra-index-url https://pypi.nvidia.com
tensorrt_cu12_libs<11
tensorrt_cu12_bindings<11; python_version < "3.13"
